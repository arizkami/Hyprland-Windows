cmake_minimum_required(VERSION 3.24)
project(HyprlandWindows VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(tiling_core src/Layout.cpp ../src/layout/algorithm/tiled/GeometryMath.cpp)
target_include_directories(tiling_core PUBLIC src)
if(BUILD_TESTING)
    add_executable(layout_tests tests/LayoutTests.cpp)
    target_link_libraries(layout_tests PRIVATE tiling_core)
    add_test(NAME layout_policy COMMAND layout_tests)
    add_executable(shared_geometry_tests tests/SharedGeometryTests.cpp)
    target_link_libraries(shared_geometry_tests PRIVATE tiling_core)
    add_test(NAME inherited_geometry_regression COMMAND shared_geometry_tests)
endif()
if(NOT WIN32)
    message(STATUS "Only portable geometry/policy tests are available on this platform")
    return()
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native milestone requires MSVC x64")
endif()
include(FetchContent)
FetchContent_Declare(yoga
    GIT_REPOSITORY https://github.com/facebook/yoga.git
    GIT_TAG 042f5013152eb81c1552dec945b88f7b95ca350f
    SOURCE_SUBDIR yoga
    GIT_PROGRESS TRUE)
FetchContent_MakeAvailable(yoga)
add_executable(HyprlandWindows WIN32 ../src/main.cpp platform/win32/Application.cpp src/Panel.cpp src/Graphics.cpp platform/win32/WindowManager.cpp app.rc)
target_compile_definitions(HyprlandWindows PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(HyprlandWindows PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_options(HyprlandWindows PRIVATE /MANIFEST:NO)
target_link_libraries(HyprlandWindows PRIVATE tiling_core yogacore d3d11 dxgi dcomp d2d1 dwrite d3dcompiler dwmapi shcore shell32)
if(BUILD_TESTING)
    add_executable(panel_tests tests/PanelTests.cpp src/Panel.cpp)
    target_compile_definitions(panel_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(panel_tests PRIVATE /utf-8)
    target_include_directories(panel_tests PRIVATE src)
    target_link_libraries(panel_tests PRIVATE yogacore)
    add_test(NAME panel_layout_clipping COMMAND panel_tests)
    add_executable(native_tests tests/NativeTests.cpp)
    target_compile_definitions(native_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(native_tests PRIVATE dwmapi shcore)
    add_test(NAME native_lifecycle COMMAND native_tests $<TARGET_FILE:HyprlandWindows>)
    set_tests_properties(native_lifecycle PROPERTIES TIMEOUT 60)
endif()
